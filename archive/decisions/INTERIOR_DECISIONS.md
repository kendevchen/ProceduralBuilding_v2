## 各階段的結果與數字

I1 結果（2026-10-02）：
- 25,920 種組合全部通過，產生加平面每棟約 1.5 ms。
- 預設建築整個重建（含零件實例化）約 4 ms。
- 截圖目視檢查了預設建築的各層、斜切轉角、連棟、街角、7 開間（兩戶加服務樓梯）、8 個側面開間（26 m 深）。

I2 結果（2026-10-02，預設建築）：

| 項目 | 結果 | 預算 |
|---|---|---|
| 室內白模 | 5,828 個三角形、4 次繪製（牆、地板、天花、樓板內的切口面） | 40,000、40 次 |
| 整棟繪製次數 | 不剖開 102 次，剖開 105 次 | |

截圖目視檢查了：

平面檢查的 2,880 種組合仍全部通過。

### 10.3 平面資料

座標和建築相同：Blender 座標，單位公尺，xy 是平面，z 朝上，原點是建築平面的左下角（`footprint` 的同一套座標）。

完整定義在 `src/plan.ts`，摘要如下：

```ts
type RoomType = "vestibule" | "concierge" | "shop" | "shopBack" | "stair" | "corridor"
  | "salon" | "dining" | "ballroom" | "study" | "bedroom" | "kitchen" | "wc" | "maid" | "storage";

interface BuildingPlan {
  width: number; length: number;
  inner: V2[];                // 外牆內面（逆時針）
  levels: PlanLevel[];        // { index, name: "1F"…"閣樓", cls: "G"|"N"|"S"|"A"|"R", floorZ, ceilingZ }
  rooms: PlanRoom[];
  walls: PlanWall[];
  windows: PlanWindow[];      // 每一層外牆上的每個開口
  stairs: { kind: "main" | "service"; polygon: V2[]; from: number; to: number; layout: StairLayout }[];
  voids: { level: number; polygon: V2[] }[];   // 樓板開口：宴會廳的上半部（樓梯井另計）
  issues: string[];           // checkPlan() 的結果
}
interface PlanRoom {
  id: string;                 // 例如 "2F-03"
  type: RoomType;
  name: string;               // 繁體中文名稱
  level: number;              // 0 = 1F；宴會廳記在下面那層
  levels: number;             // 跨幾層（宴會廳是 2）
  apartment: number | null;   // 同一層的第幾戶；樓梯、大廳、店面等公共空間是 null
  rect: [x0, y0, x1, y1];     // 牆中心線之間
  polygon: V2[];              // 房間淨空，牆面內側，逆時針
  area: number;
  floorZ: number;
  ceilingZ: number;
  windows: number[];          // 對應 windows 的索引
  doors: { wall: number; at: number; width: number; height: number; to: string }[];
}
interface PlanWindow {
  level: number;
  side: number; bay: number;  // 哪個立面的第幾個開間；-1 是斜切轉角的斜面
  kind: "window" | "door" | "shop" | "dormer";
  at: V2;                     // 開口中心，在外牆內面上
  dir: V2;                    // 沿立面方向（從外面看由左到右）
  width: number;
  room: string | null;
}
interface PlanWall {
  level: number;
  a: V2; b: V2;               // 牆中心線
  thickness: number;
  kind: "spine" | "cage" | "partition";
  rooms: [string, string | null];   // 兩側的房間；null 是宴會廳挑空
  openings: { at: number; width: number; height: number }[];   // at：門洞中心離 a 的距離
}
interface StairLayout {       // 樓梯的踏步（第 7 節）
  rect: [x0, y0, x1, y1];     // 樓梯間淨空；平台在 y0 那一側
  flight: number;             // 梯段寬
  landing: number;            // 平台深（從 y0 量）
  wellRadius: number;         // 梯井半寬，也是後端半圓的半徑
  walkOffset: number;         // 行走線離梯井的距離
  walk: number;               // 每層的行走線長度
  straight: number;           // 一側直段的長度
  flights: {                  // 每層一段
    level: number; z0: number; z1: number;   // 從哪一層出發、起訖高度
    risers: number; riser: number; going: number;
    start: number; end: number;              // 踏步在行走線上的起訖；前後是平台
  }[];
}
```
